RATES
Calcite
-start
1 REM PARM(1) = specific surface area of calcite, cm^2/mol calcite
2 REM PARM(2) = exponent for M/M0
10 si_cc = SI("Calcite")
20 IF (M <= 0 and si_cc < 0) THEN GOTO 200
30 k1 = 10^(0.198 - 444 / TK )
40 k2 = 10^(2.84 - 2177 /TK )
50 IF TC <= 25 THEN k3 = 10^(-5.86 - 317 / TK)
60 IF TC > 25 THEN k3 = 10^(-1.1 - 1737 / TK )
80 IF M0 > 0 THEN area = PARM(1)*M0*(M/M0)^PARM(2) ELSE area = PARM(1)*M
110 rate = area * (k1 * ACT("H+") + k2 * ACT("CO2") + k3 * ACT("H2O"))
120 rate = rate * (1 - 10^(2/3*si_cc))
130 moles = rate * 0.001 * TIME # convert from mmol to mol
200 SAVE moles
-end
END

SOLUTION 1
    pH    5.56
    Alkalinity    29.9 umol/l
    Ca    0.6        
    temperature    16.2
    units    mg/l
END
    
KINETICS 1
    Calcite
    -formula    Calcite 1
    -m0    1.7e-04
    -parms    4e+06    1
    -tol    1e-08
    -steps    50    100    100    100    100    100    100    100    100    100    100    100    100    100    100    100    100    100    100    100    100    250    250    250    250    250    250    250    250    250    250    500    500    500    500    500    500    500
INCREMENTAL_REACTIONS true
USE SOLUTION 1
SELECTED_OUTPUT 1
    -pH    true
    -si    CO2(g)    Calcite
    -kinetic_reactants    Calcite
END
